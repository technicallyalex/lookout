cmake_minimum_required(VERSION 3.28)
project(lookout VERSION 1.0.0 LANGUAGES CXX)
set(LOOKOUT_VERSION "${PROJECT_VERSION}-rc.2")
set(LOOKOUT_BUILD "2026-11-10-rc2")
set(CMAKE_AUTORCC ON)
if(MSVC)
 add_compile_options(/utf-8)
endif()
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg Network)
add_library(lookout MODULE src/plugin-main.cpp src/lookout.cpp src/quick-bridge.cpp src/quick-settings.cpp src/panel-visuals.cpp src/history-graphs.cpp src/card-layout.cpp src/lookout-brand.cpp src/output-bitrate.cpp src/bitrate-card.cpp src/status-icons.cpp src/replay-buffer.cpp data/branding.qrc)
target_compile_features(lookout PRIVATE cxx_std_17)
target_compile_definitions(lookout PRIVATE LOOKOUT_VERSION="${LOOKOUT_VERSION}" LOOKOUT_BUILD="${LOOKOUT_BUILD}")
target_link_libraries(lookout PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Svg Qt6::Network)
if(WIN32)
 target_link_libraries(lookout PRIVATE user32 dwmapi shell32)
endif()
set_target_properties(lookout PROPERTIES PREFIX "")
install(TARGETS lookout LIBRARY DESTINATION lookout RUNTIME DESTINATION lookout)

include(CTest)
if(BUILD_TESTING)
 add_executable(lookout-startup-tests tests/startup.cpp src/lookout.cpp src/quick-bridge.cpp src/quick-settings.cpp src/panel-visuals.cpp src/history-graphs.cpp src/card-layout.cpp src/lookout-brand.cpp src/output-bitrate.cpp src/bitrate-card.cpp src/status-icons.cpp src/replay-buffer.cpp data/branding.qrc)
 target_compile_features(lookout-startup-tests PRIVATE cxx_std_17)
 target_compile_definitions(lookout-startup-tests PRIVATE LOOKOUT_WIDGET_QA LOOKOUT_VERSION="${LOOKOUT_VERSION}" LOOKOUT_BUILD="${LOOKOUT_BUILD}")
 target_include_directories(lookout-startup-tests PRIVATE src $<TARGET_PROPERTY:OBS::obs-frontend-api,INTERFACE_INCLUDE_DIRECTORIES>)
 target_link_libraries(lookout-startup-tests PRIVATE OBS::libobs Qt6::Widgets Qt6::Svg Qt6::Network)
 if(WIN32)
  target_link_libraries(lookout-startup-tests PRIVATE user32 gdi32 dwmapi shell32)
 endif()
 add_executable(lookout-layout-tests tests/card-layout.cpp src/card-layout.cpp)
 target_compile_features(lookout-layout-tests PRIVATE cxx_std_17)
 target_include_directories(lookout-layout-tests PRIVATE src)
 target_link_libraries(lookout-layout-tests PRIVATE Qt6::Core)
 add_executable(lookout-bitrate-tests tests/output-bitrate.cpp src/output-bitrate.cpp)
 target_compile_features(lookout-bitrate-tests PRIVATE cxx_std_17)
 target_include_directories(lookout-bitrate-tests PRIVATE src)
 target_link_libraries(lookout-bitrate-tests PRIVATE Qt6::Core)
 add_test(NAME lookout-bitrate COMMAND lookout-bitrate-tests)
 add_test(NAME lookout-layout COMMAND lookout-layout-tests)
 add_test(NAME lookout-startup COMMAND lookout-startup-tests -platform minimal:enable_fonts)
 set_tests_properties(lookout-startup PROPERTIES TIMEOUT 30)
endif()
