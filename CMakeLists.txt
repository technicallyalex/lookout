cmake_minimum_required(VERSION 3.28)
project(obs-sentinel VERSION 1.0.0 LANGUAGES CXX)
set(LOOKOUT_VERSION "${PROJECT_VERSION}-rc.1")
set(LOOKOUT_BUILD "2026-11-07-rc1")
set(CMAKE_AUTORCC ON)
if(MSVC)
 add_compile_options(/utf-8)
endif()
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg Network)
add_library(obs-sentinel MODULE src/plugin-main.cpp src/sentinel.cpp src/quick-bridge.cpp src/quick-settings.cpp src/panel-visuals.cpp src/history-graphs.cpp src/card-layout.cpp src/lookout-brand.cpp src/output-bitrate.cpp src/bitrate-card.cpp src/status-icons.cpp src/replay-buffer.cpp data/branding.qrc)
target_compile_features(obs-sentinel PRIVATE cxx_std_17)
target_compile_definitions(obs-sentinel PRIVATE SENTINEL_VERSION="${LOOKOUT_VERSION}" SENTINEL_BUILD="${LOOKOUT_BUILD}")
target_link_libraries(obs-sentinel PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Svg Qt6::Network)
if(WIN32)
 target_link_libraries(obs-sentinel PRIVATE user32 dwmapi shell32)
endif()
set_target_properties(obs-sentinel PROPERTIES PREFIX "")
install(TARGETS obs-sentinel LIBRARY DESTINATION obs-sentinel RUNTIME DESTINATION obs-sentinel)

include(CTest)
if(BUILD_TESTING)
 add_executable(sentinel-startup-tests tests/startup.cpp src/sentinel.cpp src/quick-bridge.cpp src/quick-settings.cpp src/panel-visuals.cpp src/history-graphs.cpp src/card-layout.cpp src/lookout-brand.cpp src/output-bitrate.cpp src/bitrate-card.cpp src/status-icons.cpp src/replay-buffer.cpp data/branding.qrc)
 target_compile_features(sentinel-startup-tests PRIVATE cxx_std_17)
 target_compile_definitions(sentinel-startup-tests PRIVATE SENTINEL_WIDGET_QA SENTINEL_VERSION="${LOOKOUT_VERSION}" SENTINEL_BUILD="${LOOKOUT_BUILD}")

 target_link_libraries(sentinel-startup-tests PRIVATE OBS::libobs Qt6::Widgets Qt6::Svg Qt6::Network)
 if(WIN32)
  target_link_libraries(sentinel-startup-tests PRIVATE user32 gdi32 dwmapi shell32)
 endif()
 add_executable(sentinel-layout-tests tests/card-layout.cpp src/card-layout.cpp)
 target_compile_features(sentinel-layout-tests PRIVATE cxx_std_17)
 target_include_directories(sentinel-layout-tests PRIVATE src)
 target_link_libraries(sentinel-layout-tests PRIVATE Qt6::Core)
 add_executable(sentinel-bitrate-tests tests/output-bitrate.cpp src/output-bitrate.cpp)
 target_compile_features(sentinel-bitrate-tests PRIVATE cxx_std_17)
 target_include_directories(sentinel-bitrate-tests PRIVATE src)

endif()
