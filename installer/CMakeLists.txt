cmake_minimum_required(VERSION 3.28)
project(LookoutInstaller LANGUAGES CXX RC)
if(NOT EXISTS "${LOOKOUT_PAYLOAD}")
 message(FATAL_ERROR "Set LOOKOUT_PAYLOAD to the verified portable package ZIP")
endif()
file(TO_CMAKE_PATH "${LOOKOUT_PAYLOAD}" LOOKOUT_PAYLOAD)
file(SHA256 "${LOOKOUT_PAYLOAD}" PAYLOAD_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/install.ps1" SCRIPT_HASH)
configure_file(payload.rc.in payload.rc @ONLY)
add_executable(Install-Lookout WIN32 main.cpp "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
set_property(TARGET Install-Lookout PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_features(Install-Lookout PRIVATE cxx_std_17)
target_compile_definitions(Install-Lookout PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(Install-Lookout PRIVATE shell32 ole32 user32)
