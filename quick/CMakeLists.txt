cmake_minimum_required(VERSION 3.28)
project(lookout-ui VERSION 1.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Network Svg)
configure_file(lookout.rc.in lookout.rc @ONLY)
qt_add_executable(lookout-ui main.cpp backdrop.cpp backdrop.hpp panel-transition.cpp panel-transition.hpp shared-preview.cpp shared-preview.hpp ../data/branding.qrc quick.qrc "${CMAKE_CURRENT_BINARY_DIR}/lookout.rc")
target_compile_definitions(lookout-ui PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(lookout-ui PRIVATE /utf-8)
target_link_libraries(lookout-ui PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Svg d3d11 dxgi dxguid user32 dwmapi comdlg32 coremessaging windowsapp)
set_target_properties(lookout-ui PROPERTIES WIN32_EXECUTABLE TRUE)
install(TARGETS lookout-ui RUNTIME DESTINATION obs-sentinel/ui)

include(CTest)
if(BUILD_TESTING)
 add_executable(lookout-transition-tests tests/panel-transition.cpp panel-transition.cpp)
 target_include_directories(lookout-transition-tests PRIVATE .)
 target_link_libraries(lookout-transition-tests PRIVATE Qt6::Core)
 add_test(NAME lookout-transitions COMMAND lookout-transition-tests)

 find_package(Qt6 REQUIRED COMPONENTS QuickTest)
 add_executable(lookout-qml-tests ../tests/qml-smoke.cpp ../data/branding.qrc quick.qrc)
 target_link_libraries(lookout-qml-tests PRIVATE Qt6::QuickTest Qt6::QuickControls2)
 add_test(NAME lookout-qml COMMAND lookout-qml-tests -platform minimal:enable_fonts
          -input "${CMAKE_CURRENT_SOURCE_DIR}/../tests/qml" -o -,txt)
 set_tests_properties(lookout-qml PROPERTIES TIMEOUT 30
                      ENVIRONMENT "QT_QUICK_BACKEND=software;QT_FORCE_STDERR_LOGGING=1")
endif()
